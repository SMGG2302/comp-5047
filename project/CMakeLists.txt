cmake_minimum_required(VERSION 3.16)
project(hollowhaven_receiver C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
    # Produce an executable that does not require a separate VC runtime install.
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(telemetry shared/telemetry.c)
target_include_directories(telemetry PUBLIC shared)
add_executable(receiver pc_receiver/receiver.c)
target_link_libraries(receiver PRIVATE telemetry)
if(WIN32)
    target_link_libraries(receiver PRIVATE ws2_32)
    add_executable(audio_receiver pc_receiver/audio_receiver.c)
    target_include_directories(audio_receiver PRIVATE shared)
    target_link_libraries(audio_receiver PRIVATE ws2_32)
    if(MSVC)
        target_compile_options(audio_receiver PRIVATE /W4 /WX)
    endif()
else()
    target_link_libraries(telemetry PUBLIC m)
endif()
if(MSVC)
    target_compile_options(telemetry PRIVATE /W4 /WX)
    target_compile_options(receiver PRIVATE /W4 /WX)
else()
    target_compile_options(telemetry PRIVATE -Wall -Wextra -Werror)
    target_compile_options(receiver PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_executable(test_telemetry tests/test_telemetry.c)
target_link_libraries(test_telemetry PRIVATE telemetry)
add_test(NAME telemetry COMMAND test_telemetry)
add_executable(test_presence tests/test_presence.c)
target_include_directories(test_presence PRIVATE shared)
add_test(NAME presence COMMAND test_presence)

add_executable(test_moisture tests/test_moisture.c)
target_include_directories(test_moisture PRIVATE shared)
add_test(NAME moisture COMMAND test_moisture)
add_executable(test_maintenance tests/test_maintenance.c)
target_include_directories(test_maintenance PRIVATE shared)
add_test(NAME maintenance COMMAND test_maintenance)
